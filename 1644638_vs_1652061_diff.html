/**
* MakeRuns.java
*
* Creates initial runs for external sort merge using replacement selection.
* Reads from standard input and writes runs to standard output.
*
* Name: Bismi Ansar Huraisha Beevi
* ID: 1644638

import java.io.*;
import java.nio.charset.StandardCharsets;
import java.util.PriorityQueue;
public class MakeRuns {
static final int DEFAULT_H = 256;
static class Record {
String text;
String eol;
Record(String text, String eol) {
this.text = text;
this.eol = eol;
}
}
static class LineReader {
private Reader in;
private int pushed = -1;
LineReader(Reader in) {
this.in = in;
}
Record nextRecord() throws IOException {
StringBuilder text = new StringBuilder();
StringBuilder eol = new StringBuilder();
int ch = readChar();
if (ch == -1) {
return null;
}
while (ch != -1 && ch != '\n' && ch != '\r') {
text.append((char) ch);
ch = readChar();
}
if (ch == '\r') {
eol.append('\r');
int next = readChar();
if (next == '\n') {
eol.append('\n');
} else if (next != -1) {
pushed = next;
}
} else if (ch == '\n') {
eol.append('\n');
}
return new Record(text.toString(), eol.toString());
}
private int readChar() throws IOException {
if (pushed != -1) {
int ch = pushed;
pushed = -1;
return ch;
}
return in.read();
}
}
public static void main(String[] args) throws Exception {
int heapSize = getHeapSize(args);
LineReader reader = new LineReader(
new InputStreamReader(
new BufferedInputStream(System.in),
StandardCharsets.UTF_8
)
);
BufferedWriter out = new BufferedWriter(
new OutputStreamWriter(System.out, StandardCharsets.UTF_8)
);
PriorityQueue<Record> currentHeap = new PriorityQueue<>(
(a, b) -> a.text.compareTo(b.text)
);
PriorityQueue<Record> nextHeap = new PriorityQueue<>(
(a, b) -> a.text.compareTo(b.text)
);
Record deferredLastRecord = null;
// Fill initial heap
for (int i = 0; i < heapSize; i++) {
Record r = reader.nextRecord();
if (r == null) {
break;
}
// If the final input line has no newline, hold it back until the end
if (r.eol.equals("")) {
deferredLastRecord = r;
break;
}
currentHeap.add(r);
}
String lastOut = null;
while (!currentHeap.isEmpty()) {
Record smallest = currentHeap.poll();
out.write(smallest.text);
out.write(smallest.eol);
lastOut = smallest.text;
Record nextRecord = reader.nextRecord();
if (nextRecord != null) {
// If this is the final record with no newline, keep it for the end
if (nextRecord.eol.equals("")) {
deferredLastRecord = nextRecord;
} else if (nextRecord.text.compareTo(lastOut) >= 0) {
currentHeap.add(nextRecord);
} else {
nextHeap.add(nextRecord);
}
}
// Current run finished, so start the next run
if (currentHeap.isEmpty() && !nextHeap.isEmpty()) {
currentHeap = nextHeap;
nextHeap = new PriorityQueue<>(
(a, b) -> a.text.compareTo(b.text)
);
lastOut = null;
}
}
// Write the final record with no newline only at the very end
if (deferredLastRecord != null) {
out.write(deferredLastRecord.text);
out.write(deferredLastRecord.eol);
}
out.flush();
}
static int getHeapSize(String[] args) {
if (args.length != 1 || !args[0].startsWith("-h")) {
return DEFAULT_H;
}
try {
int h = Integer.parseInt(args[0].substring(2));
if (h < 32 || h > 2048) {
return DEFAULT_H;
}
return h;
} catch (NumberFormatException e) {
return DEFAULT_H;
}
}

import java.io.*;
import java.nio.charset.StandardCharsets;
import java.util.PriorityQueue;
/**
* MakeRuns.java
*
* Takes text input line by line and builds the initial sorted runs using
* replacement selection. Basically keeps a heap of size H and tries to
* squeeze as much into each run as possible before starting a new one.
* Output goes to init.runs which MergeRuns then picks up.
*
* Author: Gloria Susan Saji
* ID: 1652061
*/
public class MakeRuns {
private static final int DEFAULT_HEAP_CAPACITY = 256;
// just bundles a line with whatever line ending it came with (\n, \r\n etc.)
// so we can write it back out exactly as it was
static class Entry {
String line;
String lineEnding;
Entry(String line, String lineEnding) {
this.line = line;
this.lineEnding = lineEnding;
}
}
// reads one line at a time but saves the exact EOL characters instead of
// throwing them away like readLine() does
static class EntryReader {
private Reader source;
// needed to peek ahead one char when checking for \r\n pairs
private int buffered = -1;
EntryReader(Reader source) {
this.source = source;
}
Entry nextEntry() throws IOException {
StringBuilder lineText = new StringBuilder();
StringBuilder eol = new StringBuilder();
int ch = pull();
if (ch == -1) return null;
while (ch != -1 && ch != '\n' && ch != '\r') {
lineText.append((char) ch);
ch = pull();
}
// figure out if it's \r, \n, or \r\n and save it
if (ch == '\r') {
eol.append('\r');
int following = pull();
if (following == '\n') {
eol.append('\n');
} else if (following != -1) {
buffered = following; // not part of EOL, put it back
}
} else if (ch == '\n') {
eol.append('\n');
}
return new Entry(lineText.toString(), eol.toString());
}
private int pull() throws IOException {
if (buffered != -1) {
int ch = buffered;
buffered = -1;
return ch;
}
return source.read();
}
}
public static void main(String[] args) throws Exception {
int heapCapacity = parseHeapSize(args);
EntryReader reader = new EntryReader(
new InputStreamReader(new BufferedInputStream(System.in), StandardCharsets.UTF_8)
);
BufferedWriter writer = new BufferedWriter(
new OutputStreamWriter(System.out, StandardCharsets.UTF_8)
);
// activeHeap = lines that can still go into the current run
// deferredHeap = lines that came in too small so they have to wait for the next run
PriorityQueue<Entry> activeHeap = new PriorityQueue<>(
(x, y) -> x.line.compareTo(y.line)
);
PriorityQueue<Entry> deferredHeap = new PriorityQueue<>(
(x, y) -> x.line.compareTo(y.line)
);
// fill the heap up before we start outputting anything
while (activeHeap.size() < heapCapacity) {
Entry e = reader.nextEntry();
if (e == null) break;
activeHeap.add(e);
}
String lastWritten = null;
while (!activeHeap.isEmpty()) {
Entry smallest = activeHeap.poll();
writer.write(smallest.line);
writer.write(smallest.lineEnding);
lastWritten = smallest.line;
// grab the next line from input and decide where it goes
Entry incoming = reader.nextEntry();
if (incoming != null) {
if (incoming.line.compareTo(lastWritten) >= 0) {
activeHeap.add(incoming); // fits in current run
} else {
deferredHeap.add(incoming); // too small, save for next run
}
}
// current run is done, swap in the deferred lines for the next run
if (activeHeap.isEmpty() && !deferredHeap.isEmpty()) {
activeHeap = deferredHeap;
deferredHeap = new PriorityQueue<>((x, y) -> x.line.compareTo(y.line));
lastWritten = null;
}
}
writer.flush();
}
// pulls H out of -hH argument, falls back to default if something's wrong
static int parseHeapSize(String[] args) {
if (args.length != 1 || !args[0].startsWith("-h")) {
return DEFAULT_HEAP_CAPACITY;
}
try {
int h = Integer.parseInt(args[0].substring(2));
if (h < 32 || h > 2048) return DEFAULT_HEAP_CAPACITY;
return h;
} catch (NumberFormatException e) {
return DEFAULT_HEAP_CAPACITY;
}
}
