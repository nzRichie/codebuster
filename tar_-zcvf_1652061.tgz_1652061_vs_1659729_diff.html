/**
* Adds two number strings using long addition.
*
* @param a first number string
* @param b second number string
* @return sum of a and b as a string
* @throws IllegalArgumentException if input is null or not digits only
*/
public static String StrAdd(String a, String b) {
validateNumber(a);
validateNumber(b);
a = removeZeros(a);
b = removeZeros(b);
if (a.equals("0")) {
return b;
}
if (b.equals("0")) {
return a;
}
String answer = "";
String carry = "0";
int i = a.length() - 1;
int j = b.length() - 1;
while (i >= 0 || j >= 0) {
String leftDigit = (i >= 0) ? a.substring(i, i + 1) : "0";
String rightDigit = (j >= 0) ? b.substring(j, j + 1) : "0";
String total = addDigitsWithCarry(leftDigit, rightDigit, carry);
if (total.length() == 1) {
carry = "0";
answer = total + answer;
} else {
carry = total.substring(0, 1);
answer = total.substring(1) + answer;
}
i--;
j--;
}
if (!carry.equals("0")) {
answer = carry + answer;
}
return removeZeros(answer);
}
/**
* Multiplies two number strings using long multiplication.
*
* @param a first number string
* @param b second number string
* @return product of a and b as a string
* @throws IllegalArgumentException if input is null or not digits only
*/
public static String StrMult(String a, String b) {
validateNumber(a);
validateNumber(b);
a = removeZeros(a);
b = removeZeros(b);
if (a.equals("0") || b.equals("0")) {
return "0";
}
if (a.equals("1")) {
return b;
}
if (b.equals("1")) {
return a;
}
String result = "0";
int zerosToAdd = 0;
for (int i = b.length() - 1; i >= 0; i--) {
String digit = b.substring(i, i + 1);
String partial = multOne(a, digit);
partial = appendZeroes(partial, zerosToAdd);
result = StrAdd(result, partial);
zerosToAdd++;
}
return removeZeros(result);
}
/**
* Calculates factorial of a number string.
*
* @param a number string
* @return factorial of a as a string
* @throws IllegalArgumentException if input is null or not digits only
*/
public static String StrFact(String a) {
validateNumber(a);
a = removeZeros(a);
if (a.equals("0") || a.equals("1")) {
return "1";
}
String result = "1";
String current = a;
while (!current.equals("1")) {
result = StrMult(result, current);
current = minusOne(current);
}
return result;
}

// subtracts 1 from a number string
private static String minusOne(String s) {
if (s.equals("0")) {
return "0";
}
char[] chars = s.toCharArray();
int i = chars.length - 1;
while (i >= 0) {
String digit = String.valueOf(chars[i]);
if (!digit.equals("0")) {
chars[i] = previousDigit(digit).charAt(0);
break;
} else {
chars[i] = '9';
i--;
}
}
return removeZeros(new String(chars));
}

case "5": return "4";
case "6": return "5";
case "7": return "6";
case "8": return "7";
case "9": return "8";
default: throw new IllegalArgumentException("Invalid digit");
}
}
